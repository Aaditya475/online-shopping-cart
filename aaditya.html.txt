  <script>
    /**
     * =========================================================================
     * CLASS 1: Item
     * Represents a product or a composite bundle.
     * =========================================================================
     */
    class Item {
      constructor(id, name, price, isBundle = false, bundleDiscountPercent = 10) {
        this.id = id;
        this.name = name;
        this.price = Number(price);
        this.isBundle = Boolean(isBundle);
        this.bundleDiscountPercent = Number(bundleDiscountPercent);
      }
    }

    /**
     * =========================================================================
     * CLASS 2: Cart
     * Manages cart items, computes total with recursive discounts,
     * and returns the final bill as an object.
     * =========================================================================
     */
    class Cart {
      constructor(taxRate = 0.08) {
        this.items = []; // Array of { item: Item, quantity: number }
        this.taxRate = taxRate;
        this.bundleSize = 3;       // 3 items per bundle
        this.discountPerBundle = 15.0; // $15 off per bundle
      }

      // Add item to cart
      addItem(item, quantity = 1) {
        const qty = parseInt(quantity, 10) || 1;
        const existing = this.items.find(entry => entry.item.id === item.id);
        if (existing) {
          existing.quantity += qty;
        } else {
          this.items.push({ item, quantity: qty });
        }
      }

      // Remove item by ID
      removeItem(itemId, quantity = 1) {
        const index = this.items.findIndex(entry => entry.item.id === itemId);
        if (index !== -1) {
          this.items[index].quantity -= quantity;
          if (this.items[index].quantity <= 0) {
            this.items.splice(index, 1);
          }
        }
      }

      clear() {
        this.items = [];
      }

      /**
       * RECURSIVE FUNCTION: Compute bundle discounts
       * Recursively processes groups of 3 items from an array:
       * - Base Case: array has fewer items than bundleSize (< 3). Return 0.
       * - Recursive Case: discountPerBundle + recurse on the remaining items.
       */
      computeRecursiveBundleDiscount(itemsList) {
        // Base Case
        if (!itemsList || itemsList.length < this.bundleSize) {
          return 0;
        }
        // Recursive Step
        const remainingItems = itemsList.slice(this.bundleSize);
        return this.discountPerBundle + this.computeRecursiveBundleDiscount(remainingItems);
      }

      // Compute subtotal, recursive discounts, taxes, and final total
      computeTotal() {
        let rawSubtotal = 0;
        let itemBundleDiscount = 0;
        const expandedItems = [];

        for (const entry of this.items) {
          const lineCost = entry.item.price * entry.quantity;
          rawSubtotal += lineCost;

          // If item is marked as a bundle item, apply its specific percentage discount
          if (entry.item.isBundle) {
            itemBundleDiscount += lineCost * (entry.item.bundleDiscountPercent / 100);
          } else {
            // Expand quantity into individual items for the recursive 3-pack offer
            for (let i = 0; i < entry.quantity; i++) {
              expandedItems.push(entry.item);
            }
          }
        }

        // Call recursion to compute bundle pack discounts
        const packDiscount = this.computeRecursiveBundleDiscount(expandedItems);
        const totalDiscount = Number((itemBundleDiscount + packDiscount).toFixed(2));
        const discountedSubtotal = Math.max(0, rawSubtotal - totalDiscount);
        const tax = Number((discountedSubtotal * this.taxRate).toFixed(2));
        const finalTotal = Number((discountedSubtotal + tax).toFixed(2));

        return {
          rawSubtotal: Number(rawSubtotal.toFixed(2)),
          totalDiscount,
          discountedSubtotal: Number(discountedSubtotal.toFixed(2)),
          tax,
          finalTotal
        };
      }

      /**
       * Return the final bill as a structured JavaScript object
       */
      generateBill(customerName = "Guest Shopper") {
        const totals = this.computeTotal();

        // The final bill object
        const billObject = {
          invoiceId: "INV-" + Math.floor(100000 + Math.random() * 900000),
          date: new Date().toISOString(),
          customer: customerName,
          purchasedItems: this.items.map(entry => ({
            id: entry.item.id,
            name: entry.item.name,
            unitPrice: entry.item.price,
            quantity: entry.quantity,
            isBundle: entry.item.isBundle,
            lineTotal: Number((entry.item.price * entry.quantity).toFixed(2))
          })),
          summary: {
            rawSubtotal: totals.rawSubtotal,
            discountsApplied: totals.totalDiscount,
            discountedSubtotal: totals.discountedSubtotal,
            taxAmount: totals.tax,
            netPayableTotal: totals.finalTotal
          },
          status: "PAID"
        };

        return billObject;
      }
    }

    /**
     * =========================================================================
     * UI LOGIC & EVENT HANDLING
     * =========================================================================
     */
    const cart = new Cart(0.08);

    const itemForm = document.getElementById('itemForm');
    const cartTableBody = document.getElementById('cartTableBody');
    const txtRaw = document.getElementById('txtRaw');
    const txtDiscount = document.getElementById('txtDiscount');
    const txtTax = document.getElementById('txtTax');
    const txtTotal = document.getElementById('txtTotal');
    const btnCheckout = document.getElementById('btnCheckout');
    const outputContainer = document.getElementById('outputContainer');
    const jsonBillOutput = document.getElementById('jsonBillOutput');
    const btnClearCart = document.getElementById('btnClearCart');

    // Update the visual cart table and summary
    function renderCart() {
      cartTableBody.innerHTML = '';

      if (cart.items.length === 0) {
        cartTableBody.innerHTML = '<tr><td colspan="5" style="text-align:center; color: var(--muted);">Cart is empty</td></tr>';
      } else {
        cart.items.forEach(entry => {
          const row = document.createElement('tr');
          const lineTotal = (entry.item.price * entry.quantity).toFixed(2);
          const bundleTag = entry.item.isBundle ? ' <span class="badge">Bundle</span>' : '';

          row.innerHTML = `
            <td>${entry.item.name}${bundleTag}</td>
            <td>$${entry.item.price.toFixed(2)}</td>
            <td>${entry.quantity}</td>
            <td>$${lineTotal}</td>
            <td><button class="btn btn-danger" onclick="removeItem('${entry.item.id}')">Remove</button></td>
          `;
          cartTableBody.appendChild(row);
        });
      }

      const totals = cart.computeTotal();
      txtRaw.textContent = `$${totals.rawSubtotal.toFixed(2)}`;
      txtDiscount.textContent = `-$${totals.totalDiscount.toFixed(2)}`;
      txtTax.textContent = `$${totals.tax.toFixed(2)}`;
      txtTotal.textContent = `$${totals.finalTotal.toFixed(2)}`;
    }

    // Global helper for remove button
    window.removeItem = function(id) {
      cart.removeItem(id, 1);
      renderCart();
    };

    // Handle form submit (Add Item)
    itemForm.addEventListener('submit', (e) => {
      e.preventDefault();
      const name = document.getElementById('itemName').value.trim();
      const price = parseFloat(document.getElementById('itemPrice').value);
      const qty = parseInt(document.getElementById('itemQty').value, 10);
      const isBundle = document.getElementById('itemBundle').value === 'true';

      const id = 'ITEM_' + Date.now();
      const newItem = new Item(id, name, price, isBundle);

      cart.addItem(newItem, qty);
      renderCart();

      // Reset input fields
      document.getElementById('itemName').value = '';
      document.getElementById('itemPrice').value = '';
      document.getElementById('itemQty').value = '1';
    });

    // Clear cart
    btnClearCart.addEventListener('click', () => {
      cart.clear();
      renderCart();
      outputContainer.style.display = 'none';
    });

    // Checkout: Generate and display the Final Bill Object
    btnCheckout.addEventListener('click', () => {
      if (cart.items.length === 0) {
        alert('Cart is empty. Please add items first.');

      }

      const finalBillObject = cart.generateBill("Alice Cooper");

      // Log the actual object in the browser console
      console.log("FINAL BILL OBJECT:", finalBillObject);

      // Display the JSON formatted object on the page
      jsonBillOutput.textContent = JSON.stringify(finalBillObject, null, 2);
      outputContainer.style.display = 'block';
    });

    // Add initial demo items so the cart has content right away
    cart.addItem(new Item('1', 'USB-C Cable', 15.00), 2);
    cart.addItem(new Item('2', 'Wireless Mouse', 25.00), 1); // 2 + 1 = 3 items (triggers $15 recursive bundle discount)
    renderCart();
